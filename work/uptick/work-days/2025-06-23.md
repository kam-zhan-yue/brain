```python
def get_keys(self) -> list[FileKey]:
	keys = []
	attachments: dict = self.model_object.attachments or {}
	for filepath in attachments.values():
		# Key may be None (old correspondence data); skip these.
		if filepath is None:
			# Log if it's a recent occurence though!
			if self.model_object.created > arrow.get(2019, 1, 1):
				logger.error(
					f"Correspondence {self.model_object.pk} has attachment with"
					" missing file path."
				)
			continue
		keys.append(FileKey(filepath))
	return keys
```
